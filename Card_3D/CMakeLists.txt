cmake_minimum_required(VERSION 3.16)
project(Card3D VERSION 1.0 LANGUAGES CXX)
add_library(card3d MODULE src/card_studio.cpp)
target_compile_features(card3d PRIVATE cxx_std_17)
set_target_properties(card3d PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(card3d PRIVATE /O2 /W4 /utf-8)
else()
  target_compile_options(card3d PRIVATE -O3 -Wall -Wextra -Wpedantic)
  target_link_libraries(card3d PRIVATE m)
endif()
install(TARGETS card3d LIBRARY DESTINATION lib/frei0r-1)
option(CARD3D_STUDIO_RESOURCES "Install native Studio resources from prepared package" OFF)
if(CARD3D_STUDIO_RESOURCES)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/previews/previews.json")
    message(FATAL_ERROR "Generate and verify Studio previews before packaging")
  endif()
  install(FILES kdenlive/card3d.xml DESTINATION share/kdenlive/effects)
  install(FILES kdenlive/studio.json DESTINATION share/kdenlive/studio)
  install(DIRECTORY previews/ DESTINATION share/kdenlive/studio)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(card-memory-buffers tests/memory_buffers.cpp)
  target_compile_features(card-memory-buffers PRIVATE cxx_std_17)
  target_compile_options(card-memory-buffers PRIVATE -UNDEBUG)
  add_test(NAME card-memory-buffers COMMAND card-memory-buffers)
  set_tests_properties(card-memory-buffers PROPERTIES TIMEOUT 20)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(card3d_test MODULE src/card_studio.cpp)
  set_target_properties(card3d_test PROPERTIES PREFIX "")
  target_compile_features(card3d_test PRIVATE cxx_std_17)
  target_compile_definitions(card3d_test PRIVATE CARD3D_TEST_API)
  if(NOT MSVC)
    target_compile_options(card3d_test PRIVATE -O2 -Wall -Wextra -Wpedantic)
    target_link_libraries(card3d_test PRIVATE m)
  endif()
  foreach(suite engine public geometry)
    add_test(NAME card-${suite} COMMAND ${CMAKE_COMMAND} -E env
      CARD3D_LIBRARY=$<TARGET_FILE:card3d> CARD3D_TEST_LIBRARY=$<TARGET_FILE:card3d_test>
      CARD3D_GEOMETRY_REPORT=${CMAKE_CURRENT_BINARY_DIR}/geometry-results.json
      PYTHONDONTWRITEBYTECODE=1
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${suite}.py)
    set_tests_properties(card-${suite} PROPERTIES TIMEOUT 120)
  endforeach()
endif()
