cmake_minimum_required(VERSION 3.16)
project(StudioCamera C CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(STUDIO_CAMERA_MLT "Build the MLT module (official headers required)" ON)
option(STUDIO_CAMERA_HEAD_TRACKER "Build the native offline head analyser" ON)
add_library(studio_camera_core STATIC src/camera.c src/track.c)
set_target_properties(studio_camera_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(studio_camera_core PRIVATE -O3 -Wall -Wextra -fno-math-errno)
target_link_libraries(studio_camera_core PUBLIC m pthread)
if(STUDIO_CAMERA_MLT)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(MLT REQUIRED IMPORTED_TARGET mlt-framework-7)
  add_library(mltstudiocamera MODULE src/mlt_camera.c)
  set_target_properties(mltstudiocamera PROPERTIES PREFIX "lib")
  target_compile_options(mltstudiocamera PRIVATE -O3 -Wall -Wextra -fno-math-errno)
  target_link_libraries(mltstudiocamera PRIVATE studio_camera_core PkgConfig::MLT)
  install(TARGETS mltstudiocamera LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/mlt-7)
  install(FILES mlt/filter_camera.yml DESTINATION share/mlt-7/studio)
  install(FILES kdenlive/studio_camera.xml DESTINATION share/kdenlive/effects)
endif()
if(STUDIO_CAMERA_HEAD_TRACKER)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio tracking)
  add_executable(studio-head-tracker src/head_tracker.cpp)
  target_compile_options(studio-head-tracker PRIVATE -O3 -Wall -Wextra)
  target_link_libraries(studio-head-tracker PRIVATE ${OpenCV_LIBS})
  install(TARGETS studio-head-tracker RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
  add_executable(camera-engine-tests tests/test_engine.c)
  add_executable(camera-track-tests tests/test_track.c)
  foreach(test_target camera-engine-tests camera-track-tests)
    target_link_libraries(${test_target} PRIVATE studio_camera_core)
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
  endforeach()
  add_test(NAME camera-engine COMMAND camera-engine-tests)
  add_test(NAME camera-track COMMAND camera-track-tests ${CMAKE_CURRENT_SOURCE_DIR}/tests/sample.scam)
  set_tests_properties(camera-engine camera-track PROPERTIES TIMEOUT 30)
endif()
install(FILES kdenlive/studio_camera.json DESTINATION share/kdenlive/studio)
