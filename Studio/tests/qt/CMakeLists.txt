cmake_minimum_required(VERSION 3.20)
project(StudioQtBoundaryTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(studio-resource-tests test_resources.cpp)
target_link_libraries(studio-resource-tests PRIVATE Qt6::Core)
target_compile_options(studio-resource-tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME studio-resources COMMAND studio-resource-tests)
set_tests_properties(studio-resources PROPERTIES TIMEOUT 20)

find_package(Qt6 REQUIRED COMPONENTS Gui Xml)
add_executable(studio-mask-analysis-tests test_mask_analysis.cpp)
target_link_libraries(studio-mask-analysis-tests PRIVATE Qt6::Core Qt6::Gui Qt6::Xml)
target_compile_options(studio-mask-analysis-tests PRIVATE -UNDEBUG)
add_test(NAME studio-mask-analysis COMMAND studio-mask-analysis-tests)
set_tests_properties(studio-mask-analysis PROPERTIES TIMEOUT 20)
