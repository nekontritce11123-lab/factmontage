cmake_minimum_required(VERSION 3.18)
project(StudioTransitions VERSION 1.0 LANGUAGES CXX)

include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(TRANSITIONS_SYSTEM_FREI0R "Build against the SDK frei0r.h" OFF)
option(TRANSITIONS_SCALAR_REFERENCE "Disable the SSE2 fast path" OFF)

add_library(sunimo_transition MODULE src/plugin_transition.cpp src/transition.cpp)
set_target_properties(sunimo_transition PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
target_include_directories(sunimo_transition PRIVATE src)
if(TRANSITIONS_SYSTEM_FREI0R)
  find_path(FREI0R_INCLUDE frei0r.h REQUIRED)
  target_include_directories(sunimo_transition PRIVATE ${FREI0R_INCLUDE})
  target_compile_definitions(sunimo_transition PRIVATE TRANSITIONS_SYSTEM_FREI0R)
endif()
if(TRANSITIONS_SCALAR_REFERENCE)
  target_compile_definitions(sunimo_transition PRIVATE TRANSITIONS_SCALAR_REFERENCE)
endif()
if(MSVC)
  target_compile_options(sunimo_transition PRIVATE /O2 /W4 /utf-8)
else()
  target_compile_options(sunimo_transition PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
  add_executable(transition-tests tests/transition_tests.cpp src/transition.cpp)
  target_include_directories(transition-tests PRIVATE src)
  if(NOT MSVC)
    target_compile_options(transition-tests PRIVATE -UNDEBUG)
  endif()
  if(MSVC)
    target_compile_options(transition-tests PRIVATE /W4 /utf-8)
  else()
    target_compile_options(transition-tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME transition-core COMMAND transition-tests)

  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_library(transition_scalar MODULE src/plugin_transition.cpp src/transition.cpp)
  target_include_directories(transition_scalar PRIVATE src)
  target_compile_definitions(transition_scalar PRIVATE TRANSITIONS_SCALAR_REFERENCE)
  add_test(NAME transition-renderer-parity COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/compare_renderers.py
    $<TARGET_FILE:sunimo_transition> $<TARGET_FILE:transition_scalar>)
  set_tests_properties(transition-renderer-parity PROPERTIES TIMEOUT 120)

  find_path(FREI0R_TEST_INCLUDE frei0r.h)
  if(FREI0R_TEST_INCLUDE)
    add_executable(transition-abi tests/abi_host.cpp)
    target_include_directories(transition-abi PRIVATE ${FREI0R_TEST_INCLUDE})
    target_link_libraries(transition-abi PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME transition-abi COMMAND transition-abi $<TARGET_FILE:sunimo_transition>)
  else()
    message(STATUS "Official frei0r.h absent: ABI host is deferred to the Flatpak SDK")
  endif()
endif()

install(TARGETS sunimo_transition LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/frei0r-1)
install(FILES kdenlive/studio_transition.xml DESTINATION share/kdenlive/transitions)
install(FILES kdenlive/studio_transition.json kdenlive/transition_previews.json DESTINATION share/kdenlive/studio)
install(FILES src/generated_parameters.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/studio-transitions)
install(DIRECTORY sfx/ DESTINATION share/kdenlive/studio/sfx/v1 FILES_MATCHING PATTERN "transition_*.wav")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/previews/transition_previews.json")
  install(DIRECTORY previews/ DESTINATION share/kdenlive/studio FILES_MATCHING PATTERN "transition_*")
endif()
install(FILES LICENSE NOTICE DESTINATION share/licenses/studio-transitions)
