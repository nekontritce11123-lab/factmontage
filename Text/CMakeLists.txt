cmake_minimum_required(VERSION 3.16)
project(sunimo_text_studio VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(sunimo_text_studio SHARED src/plugin.cpp)
set_target_properties(sunimo_text_studio PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_compile_options(sunimo_text_studio PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_link_options(sunimo_text_studio PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
endif()
install(TARGETS sunimo_text_studio LIBRARY DESTINATION lib/frei0r-1)
install(FILES web/catalog.json DESTINATION share/studio-text)
install(FILES kdenlive/sunimo_text_studio.xml DESTINATION share/kdenlive/effects)
if(BUILD_TESTING)
  add_executable(text-memory-buffers tests/memory_buffers.cpp)
  target_include_directories(text-memory-buffers PRIVATE src)
  target_compile_options(text-memory-buffers PRIVATE -UNDEBUG)
  add_test(NAME text-memory-buffers COMMAND text-memory-buffers ${CMAKE_CURRENT_SOURCE_DIR}/examples/life_v02.stxt)
  set_tests_properties(text-memory-buffers PROPERTIES TIMEOUT 20)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(text-motion-invariants tests/motion_invariants.cpp)
  target_include_directories(text-motion-invariants PRIVATE src)
  target_compile_options(text-motion-invariants PRIVATE -UNDEBUG)
  add_test(NAME text-motion-invariants COMMAND text-motion-invariants ${CMAKE_CURRENT_SOURCE_DIR}/examples/life_v02.stxt)
  foreach(suite engine lifecycle)
    add_test(NAME text-${suite} COMMAND ${CMAKE_COMMAND} -E env
      SUNIMO_TEXT_LIBRARY=$<TARGET_FILE:sunimo_text_studio>
      SUNIMO_TEXT_INVARIANTS=$<TARGET_FILE:text-motion-invariants>
      PYTHONDONTWRITEBYTECODE=1
      ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${suite}.py)
  endforeach()
  add_test(NAME text-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_contract.py)
  set_tests_properties(text-engine text-lifecycle text-motion-invariants text-contract PROPERTIES TIMEOUT 120)
endif()

# Opt-in compiler: the render plugin itself stays free of Qt/font dependencies.
option(STXT_BUILD_QT_COMPILER "Build native Qt scene compiler and shaping tests" OFF)
if(STXT_BUILD_QT_COMPILER)
  find_package(Qt6 REQUIRED COMPONENTS Core Gui)
  add_executable(studio-text-compile qt/compile_main.cpp)
  target_link_libraries(studio-text-compile PRIVATE Qt6::Core Qt6::Gui)
  install(TARGETS studio-text-compile RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    add_executable(text-qt-compiler-test tests/test_qt_compiler.cpp)
    target_include_directories(text-qt-compiler-test PRIVATE qt src)
    target_link_libraries(text-qt-compiler-test PRIVATE Qt6::Core Qt6::Gui)
    target_compile_options(text-qt-compiler-test PRIVATE -UNDEBUG)
    add_test(NAME text-qt-compiler COMMAND text-qt-compiler-test)
    set_tests_properties(text-qt-compiler PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  endif()
endif()
