cmake_minimum_required(VERSION 3.20)
project(StudioBackground VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(SBG_WITH_MLT "Build the real MLT 7 module (requires official SDK headers)" OFF)
option(SBG_WITH_ONNX "Build native human segmentation analyzer" OFF)
option(SBG_SANITIZE "ASan + UBSan for native tests" OFF)
find_package(ZLIB REQUIRED)
find_package(PNG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_target(sbg-contract ALL COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_ui.py --check)
add_library(sbg-core STATIC src/core.cpp src/cache.cpp src/png_io.cpp src/hash.cpp)
target_include_directories(sbg-core PUBLIC include)
target_link_libraries(sbg-core PUBLIC ZLIB::ZLIB PNG::PNG OpenSSL::Crypto)
target_compile_definitions(sbg-core PRIVATE _GNU_SOURCE)
target_compile_options(sbg-core PRIVATE -Wall -Wextra -Wpedantic)
if(SBG_SANITIZE)
 target_compile_options(sbg-core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(sbg-core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sbg-cli bin/cli.cpp)
target_link_libraries(sbg-cli PRIVATE sbg-core)
add_executable(sbg-tests tests/test_core.cpp)
target_link_libraries(sbg-tests PRIVATE sbg-core)
target_compile_options(sbg-tests PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME core-cache-render COMMAND sbg-tests)
add_test(NAME generated-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_ui.py --check)
add_test(NAME contract-shape COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_contract.py)
if(SBG_WITH_MLT)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(MLT REQUIRED IMPORTED_TARGET mlt-framework-7)
 add_library(mltstudio_background MODULE mlt/filter_background.cpp)
 target_link_libraries(mltstudio_background PRIVATE sbg-core PkgConfig::MLT)
 add_executable(sbg-mlt-callback-test tests/test_mlt_callback.cpp)
 target_link_libraries(sbg-mlt-callback-test PRIVATE sbg-core PkgConfig::MLT)
 target_compile_options(sbg-mlt-callback-test PRIVATE -UNDEBUG)
 add_test(NAME mlt-background-callback COMMAND sbg-mlt-callback-test)
 set_tests_properties(mlt-background-callback PROPERTIES TIMEOUT 30)
 install(TARGETS mltstudio_background LIBRARY DESTINATION lib/mlt-7)
 install(FILES kdenlive/studio_background.xml DESTINATION share/kdenlive/effects)
endif()
if(SBG_WITH_ONNX)
 set(ONNXRUNTIME_ROOT "" CACHE PATH "Pinned native ONNX Runtime distribution")
 set(SBG_MODEL_FILE "" CACHE FILEPATH "Pinned PP-HumanSegV2-Lite portrait ONNX model")
 set(SBG_MODEL_SHA256 "2047b95bea344af15575f4279d6fac44bb07a42e7d537cf8d3265f15fe95cc82" CACHE STRING "Expected PP-HumanSegV2-Lite SHA256")
 if(NOT EXISTS "${SBG_MODEL_FILE}")
  message(FATAL_ERROR "SBG_MODEL_FILE must point to the pinned PP-HumanSegV2-Lite portrait ONNX model")
 endif()
 file(SHA256 "${SBG_MODEL_FILE}" SBG_MODEL_ACTUAL_SHA256)
 if(NOT SBG_MODEL_ACTUAL_SHA256 STREQUAL SBG_MODEL_SHA256)
  message(FATAL_ERROR "PP-HumanSegV2-Lite model SHA256 mismatch")
 endif()
 find_path(ONNX_INCLUDE onnxruntime_cxx_api.h PATHS ${ONNXRUNTIME_ROOT}/include PATH_SUFFIXES onnxruntime onnxruntime/core/session REQUIRED)
 find_library(ONNX_LIBRARY onnxruntime PATHS ${ONNXRUNTIME_ROOT}/lib REQUIRED)
 add_executable(studio-background-analyzer bin/analyze.cpp)
 target_include_directories(studio-background-analyzer PRIVATE ${ONNX_INCLUDE})
 target_link_libraries(studio-background-analyzer PRIVATE sbg-core ${ONNX_LIBRARY})
 add_test(NAME analyzer-pipe-count COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_analyzer.py
          $<TARGET_FILE:studio-background-analyzer> ${SBG_MODEL_FILE} ${SBG_MODEL_SHA256})
 set_tests_properties(analyzer-pipe-count PROPERTIES TIMEOUT 60)
 install(TARGETS studio-background-analyzer RUNTIME DESTINATION bin)
 install(FILES "${SBG_MODEL_FILE}" DESTINATION share/studio-background/models)
endif()
install(FILES kdenlive/studio_background.json DESTINATION share/kdenlive/studio)
install(DIRECTORY previews/ DESTINATION share/kdenlive/studio/previews FILES_MATCHING PATTERN "*.png" PATTERN "*.gif")
install(FILES LICENSE THIRD_PARTY_NOTICES.md licenses/ONNX_RUNTIME_LICENSE.txt DESTINATION share/licenses/studio-background)
install(FILES licenses/PADDLESEG_LICENSE.txt DESTINATION share/licenses/studio-background)
add_executable(sbg-benchmark EXCLUDE_FROM_ALL tests/benchmark.cpp)
target_link_libraries(sbg-benchmark PRIVATE sbg-core)
add_test(NAME cli-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:sbg-cli>)
