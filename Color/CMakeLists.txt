cmake_minimum_required(VERSION 3.20)
project(StudioColor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(STUDIO_COLOR_MLT "Build the MLT 7 service (requires actual MLT SDK)" OFF)
option(STUDIO_COLOR_SANITIZE "Enable address/undefined-behaviour sanitizers" OFF)
option(STUDIO_COLOR_NATIVE "Benchmark this host only; NEVER use in a Flatpak release" OFF)
option(STUDIO_COLOR_DEVEL "Install static core and development headers" ON)
find_package(Threads REQUIRED)
if(STUDIO_COLOR_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
if(STUDIO_COLOR_NATIVE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-march=native)
endif()
add_library(studio_color_core STATIC src/color.cpp src/analysis.cpp src/kernel_avx2.cpp)
target_include_directories(studio_color_core PUBLIC include)
target_link_libraries(studio_color_core PUBLIC Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(studio_color_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_executable(studio-color src/cli.cpp)
target_link_libraries(studio-color PRIVATE studio_color_core)
add_executable(color-tests tests/core_tests.cpp)
target_link_libraries(color-tests PRIVATE studio_color_core)
if(NOT MSVC)
 target_compile_options(color-tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME core COMMAND color-tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME generated-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/generate_ui.py --check)
 add_test(NAME cli-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:studio-color>)
endif()
include(GNUInstallDirs)
if(STUDIO_COLOR_DEVEL)
 install(TARGETS studio_color_core ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
 install(DIRECTORY include/studio_color DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
if(STUDIO_COLOR_MLT)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(MLT REQUIRED IMPORTED_TARGET mlt-framework-7)
 add_library(mltstudiocolor MODULE mlt/filter_studio_color.cpp)
 target_link_libraries(mltstudiocolor PRIVATE studio_color_core PkgConfig::MLT)
 target_compile_definitions(mltstudiocolor PRIVATE STUDIO_COLOR_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/mlt-7/studiocolor")
 set_target_properties(mltstudiocolor PROPERTIES PREFIX "lib")
 add_executable(color-mlt-smoke tests/mlt_smoke.cpp)
 target_link_libraries(color-mlt-smoke PRIVATE studio_color_core PkgConfig::MLT)
 add_test(NAME actual-mlt COMMAND color-mlt-smoke $<TARGET_FILE_DIR:mltstudiocolor>)
 install(TARGETS mltstudiocolor LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/mlt-7)
 install(FILES kdenlive/studio_color.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/kdenlive/effects)
 install(FILES kdenlive/studio_color.json DESTINATION ${CMAKE_INSTALL_DATADIR}/kdenlive/studio)
 install(FILES kdenlive/filter_studio_color.yml DESTINATION ${CMAKE_INSTALL_DATADIR}/mlt-7/studiocolor)
endif()
