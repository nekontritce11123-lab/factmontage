cmake_minimum_required(VERSION 3.16)
project(StudioFX VERSION 0.2.0 LANGUAGES C)
include(GNUInstallDirs)
include(CTest)
set(FREI0R_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Directory containing frei0r.h; release validation must use untouched upstream header")
if(NOT EXISTS "${FREI0R_INCLUDE_DIR}/frei0r.h")
 message(FATAL_ERROR "frei0r.h is missing from FREI0R_INCLUDE_DIR")
endif()
option(SFX_SANITIZE "Build with ASan + UBSan (validation only)" OFF)
add_library(studiofx MODULE src/studiofx.c)
add_library(studio_lens MODULE src/specialized.c)
add_library(studio_vintage MODULE src/specialized.c)
set_target_properties(studiofx PROPERTIES PREFIX "" C_STANDARD 99 C_STANDARD_REQUIRED ON C_VISIBILITY_PRESET hidden)
set_target_properties(studio_lens studio_vintage PROPERTIES PREFIX "" C_STANDARD 99 C_STANDARD_REQUIRED ON C_VISIBILITY_PRESET hidden)
target_include_directories(studiofx PRIVATE include ${FREI0R_INCLUDE_DIR})
target_include_directories(studio_lens PRIVATE include ${FREI0R_INCLUDE_DIR})
target_include_directories(studio_vintage PRIVATE include ${FREI0R_INCLUDE_DIR})
target_compile_definitions(studio_lens PRIVATE STUDIO_LENS=1)
target_compile_definitions(studio_vintage PRIVATE STUDIO_VINTAGE=1)
target_compile_options(studiofx PRIVATE -Wall -Wextra -Wpedantic -fno-math-errno)
target_compile_options(studio_lens PRIVATE -Wall -Wextra -Wpedantic -fno-math-errno)
target_compile_options(studio_vintage PRIVATE -Wall -Wextra -Wpedantic -fno-math-errno)
target_link_libraries(studiofx PRIVATE m)
target_link_libraries(studio_lens PRIVATE m)
target_link_libraries(studio_vintage PRIVATE m)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_compile_options(studiofx PRIVATE -march=x86-64 -mtune=generic)
  target_compile_options(studio_lens PRIVATE -march=x86-64 -mtune=generic)
  target_compile_options(studio_vintage PRIVATE -march=x86-64 -mtune=generic)
endif()
if(SFX_SANITIZE)
  target_compile_options(studiofx PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(studiofx PRIVATE -fsanitize=address,undefined)
  target_compile_options(studio_lens PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(studio_lens PRIVATE -fsanitize=address,undefined)
  target_compile_options(studio_vintage PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(studio_vintage PRIVATE -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
  add_executable(sfx-memory-buffers tests/memory_buffers.c)
  target_include_directories(sfx-memory-buffers PRIVATE include ${FREI0R_INCLUDE_DIR})
  target_compile_options(sfx-memory-buffers PRIVATE -UNDEBUG)
  target_link_libraries(sfx-memory-buffers PRIVATE m)
  add_test(NAME sfx-memory-buffers COMMAND sfx-memory-buffers)
  set_tests_properties(sfx-memory-buffers PROPERTIES TIMEOUT 20)
  add_executable(sfx_abi_host tests/abi_host.c)
  target_compile_options(sfx_abi_host PRIVATE -UNDEBUG)
  target_include_directories(sfx_abi_host PRIVATE ${FREI0R_INCLUDE_DIR} include)
  target_link_libraries(sfx_abi_host PRIVATE dl m)
  if(SFX_SANITIZE)
    target_compile_options(sfx_abi_host PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sfx_abi_host PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME abi_and_render COMMAND sfx_abi_host $<TARGET_FILE:studiofx>)
  add_executable(sfx_specialized_host tests/specialized_host.c)
  target_compile_options(sfx_specialized_host PRIVATE -UNDEBUG)
  target_include_directories(sfx_specialized_host PRIVATE ${FREI0R_INCLUDE_DIR})
  target_link_libraries(sfx_specialized_host PRIVATE dl m)
  add_test(NAME lens_abi_and_render COMMAND sfx_specialized_host $<TARGET_FILE:studio_lens> 16)
  add_test(NAME vintage_abi_and_render COMMAND sfx_specialized_host $<TARGET_FILE:studio_vintage> 17)
endif()
install(TARGETS studiofx studio_lens studio_vintage LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/frei0r-1)
install(FILES kdenlive/studiofx.xml kdenlive/studio_lens.xml kdenlive/studio_vintage.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/kdenlive/effects)
install(FILES kdenlive/catalog.json kdenlive/studiofx.json kdenlive/studio_lens.json kdenlive/studio_vintage.json kdenlive/specialized.json DESTINATION ${CMAKE_INSTALL_DATADIR}/kdenlive/studio)
install(DIRECTORY previews/ DESTINATION ${CMAKE_INSTALL_DATADIR}/kdenlive/studio/studiofx FILES_MATCHING PATTERN "*.png" PATTERN "*.gif")
