cmake_minimum_required(VERSION 3.20)
project(StudioSubtitles VERSION 0.6.0 LANGUAGES CXX)

if(NOT UNIX)
  message(FATAL_ERROR "Worker is POSIX-only. Build on Linux/WSL or inside the pinned Flatpak SDK.")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STUDIO_SUBTITLES_SANITIZERS "Enable address/undefined sanitizers" OFF)
set(STUDIO_SUBTITLE_MODEL_FILE "" CACHE FILEPATH "Pinned multilingual whisper.cpp model")
set(STUDIO_SUBTITLE_VAD_MODEL_FILE "" CACHE FILEPATH "Pinned whisper.cpp Silero VAD model")

if(NOT TARGET studio_audio)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/Audio/CMakeLists.txt)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/Audio ${CMAKE_CURRENT_BINARY_DIR}/studio_audio)
  else()
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../Audio ${CMAKE_CURRENT_BINARY_DIR}/studio_audio)
  endif()
endif()

add_library(studio_subtitles STATIC src/subtitle.cpp)
target_include_directories(studio_subtitles PUBLIC include PRIVATE generated)
target_link_libraries(studio_subtitles PUBLIC studio_audio)
target_compile_options(studio_subtitles PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

if(STUDIO_SUBTITLES_SANITIZERS)
  target_compile_options(studio_subtitles PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(studio_subtitles PUBLIC -fsanitize=address,undefined)
endif()

add_executable(studio-subtitle tools/studio_subtitle_cli.cpp)
target_link_libraries(studio-subtitle PRIVATE studio_subtitles)

include(CTest)
if(BUILD_TESTING)
  add_executable(subtitle-unit tests/unit.cpp)
  target_link_libraries(subtitle-unit PRIVATE studio_subtitles)
  target_compile_options(subtitle-unit PRIVATE -UNDEBUG)
  add_test(NAME subtitle-unit COMMAND subtitle-unit)
  add_test(NAME subtitle-worker-cache COMMAND /bin/sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/cache_smoke.sh
           $<TARGET_FILE:studio-subtitle> ${CMAKE_CURRENT_SOURCE_DIR}/tests/fake_whisper_cli.sh)
endif()

install(TARGETS studio-subtitle RUNTIME DESTINATION libexec)
install(FILES generated/parameters.json DESTINATION share/studio-subtitles)
install(FILES LICENSE DESTINATION share/licenses/studio-subtitles)
install(FILES licenses/WHISPER_MODEL_LICENSE.txt DESTINATION share/licenses/studio-subtitles)
install(FILES licenses/SILERO_VAD_LICENSE.txt DESTINATION share/licenses/studio-subtitles)
if(STUDIO_SUBTITLE_MODEL_FILE)
  install(FILES ${STUDIO_SUBTITLE_MODEL_FILE} DESTINATION share/studio-subtitles)
endif()
if(STUDIO_SUBTITLE_VAD_MODEL_FILE)
  install(FILES ${STUDIO_SUBTITLE_VAD_MODEL_FILE} DESTINATION share/studio-subtitles)
endif()
