cmake_minimum_required(VERSION 3.20)
project(StudioAudio VERSION 0.2.0 LANGUAGES CXX)
if(NOT UNIX)
 message(FATAL_ERROR "Worker is POSIX-only. Build on Linux/WSL or inside the pinned Flatpak SDK.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STUDIO_AUDIO_SANITIZERS "Enable address/undefined sanitizers" OFF)
set(AUDIO_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
add_library(studio_audio STATIC ${AUDIO_ROOT}/src/audio.cpp ${AUDIO_ROOT}/src/process_posix.cpp ${AUDIO_ROOT}/src/sha256.cpp)
target_include_directories(studio_audio PUBLIC ${AUDIO_ROOT}/include ${AUDIO_ROOT}/generated)
target_compile_options(studio_audio PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(STUDIO_AUDIO_SANITIZERS)
 target_compile_options(studio_audio PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(studio_audio PUBLIC -fsanitize=address,undefined)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 add_executable(studio-audio tools/studio_audio_cli.cpp)
 target_link_libraries(studio-audio PRIVATE studio_audio)
 include(CTest)
 if(BUILD_TESTING)
  add_executable(audio-unit tests/unit.cpp)
  target_link_libraries(audio-unit PRIVATE studio_audio)
  target_compile_options(audio-unit PRIVATE -UNDEBUG)
  add_test(NAME audio-unit COMMAND audio-unit)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
   add_test(NAME generated-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_ui.py --check)
  endif()
 endif()
 install(TARGETS studio-audio RUNTIME DESTINATION libexec)
 install(FILES ${AUDIO_ROOT}/generated/parameters.json DESTINATION share/studio-audio)
endif()
